\begin{tikzpicture}[
    node distance=0.5cm and 0.5cm,
    % concrete styles built on the library components
    sbox/.append style={fill=typeone, thick, minimum size=0.5cm},
    % layer={level}{label}: white full-width layer aligned with the S-box edges
    layer/.style 2 args={
        layerbox={($(x1 |- #1)+(-0.25,0)$)($(x5 |- #1)+(0.25,0)$)}{#2},
        fill=white, thick, minimum height=0.5cm},
]
    % ===== state columns (col 3 is the dots column) =====
    \node[state] (x1) {$x_1$};
    \node[state, right=of x1] (x2) {$x_2$};
    \node[state, right=of x2] (x3) {\ldots};
    \node[state, right=of x3] (x4) {$x_{t-1}$};
    \node[state, right=of x4] (x5) {$x_t$};

    % ===== vertical layout (tunable) =====
    \def\rowstep{0.7cm} % gap between adjacent layers inside a round block
    \def\blockgap{1.2cm} % gap between round blocks (full <-> partial spacing)
    \def\outgap{0.9cm}   % gap from the last matrix to the output row

    \coordinate[below=of x1] (a1); % first AddRC below the inputs
    \coordinate[below=\rowstep of a1] (s1);
    \coordinate[below=\rowstep of s1] (m1);

    \coordinate[below=\blockgap of m1] (a2);
    \coordinate[below=\rowstep of a2] (s2);
    \coordinate[below=\rowstep of s2] (m2);

    \coordinate[below=\blockgap of m2] (a3);
    \coordinate[below=\rowstep of a3] (s3);
    \coordinate[below=\rowstep of s3] (m3);

    \coordinate[below=\outgap of m3] (out);

    \coordinate (d1)  at ($(m1)!0.5!(a2)$);      % vdots centred between blocks
    \coordinate (d2)  at ($(m2)!0.5!(a3)$);

    % ===== WIRES (drawn first, covered by the white layers below) =====
    % one straight arrow per block per column, skipping the dots column
    \foreach \i in {1,2,4,5} {
        \draw[wire] (x\i.south)              -- ($(x\i |- m1)+(0,-0.55)$); % block 1 (Rf)
        \draw[wire] ($(x\i |- a2)+(0,0.45)$) -- ($(x\i |- m2)+(0,-0.55)$); % block 2 (RP)
        \draw[wire] ($(x\i |- a3)+(0,0.45)$) -- ($(x\i |- out)+(0,0.25)$); % block 3 (Rf)
    }

    % ===== LAYERS + S-BOXES (cover the wires) =====
    % block 1 (Rf): AddRC, full S-box row, M
    \node[layer={a1}{\AddRC}] {};
    \foreach \i in {1,2,4,5} {\node[sbox] at (x\i |- s1) {$x^\alpha$};}
    \node[empty] at (x3 |- s1) {\(\dots\)};
    \node[layer={m1}{$\M$}] {};

    \node[empty] at (x3 |- d1) {\(\vdots\)};

    % block 2 (RP): AddRC, single S-box (last column), M
    \node[layer={a2}{\AddRC}] {};
    \node[sbox]  at (x5 |- s2) {$x^\alpha$};
    \node[empty] at (x3 |- s2) {\(\dots\)};
    \node[layer={m2}{$\M$}] {};

    \node[empty] at (x3 |- d2) {\(\vdots\)};

    % block 3 (Rf): AddRC, full S-box row, M
    \node[layer={a3}{\AddRC}] {};
    \foreach \i in {1,2,4,5} {\node[sbox] at (x\i |- s3) {$x^\alpha$};}
    \node[empty] at (x3 |- s3) {\(\dots\)};
    \node[layer={m3}{$\M$}] {};

    % ===== output state =====
    \node[state] at (x1 |- out) {$y_1$};
    \node[state] at (x2 |- out) {$y_2$};
    \node[state] at (x3 |- out) {\ldots};
    \node[state] at (x4 |- out) {$y_{t-1}$};
    \node[state] at (x5 |- out) {$y_t$};

    % ===== round-count braces (right side) =====
    % brace drawn top->bottom (over the box span); label centred beside it
    \foreach \top/\bot/\lab in {a1/m1/\Rf, a2/m2/\RP, a3/m3/\Rf} {
        \vbrace[raise=15pt, above=-0.25cm, below=-0.25cm]{x5 |- \bot}{x5 |- \top}{x5};
        \node[gray, rotate=-90] at ($(x5 |- \top)!0.5!(x5 |- \bot)+(0.75,0)$)
            {\footnotesize $\lab$ rounds};
    }

\end{tikzpicture}
