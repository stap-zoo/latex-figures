\begin{tikzpicture}[
    node distance=0.5cm and 0.5cm,
    % concrete styles built on the library components
    sbox/.append style={fill=typeone, thick, minimum size=0.5cm},
    sboxwide/.style={bbox={1.5cm}{0.5cm}, fill=typeone!70, thick},
    % layer={level}{label}: white full-width layer aligned with the S-box edges
    layer/.style 2 args={
        layerbox={($(x1 |- #1)+(-0.25,0)$)($(x5 |- #1)+(0.25,0)$)}{#2},
        fill=white, thick, minimum height=0.5cm},
]
    % ===== state columns (col 3 is the dots column) =====
    \node[state] (x1) {$x_1$};
    \node[state, right=of x1] (x2) {$x_2$};
    \node[state, right=of x2] (x3) {\ldots};
    \node[state, right=of x3] (x4) {$x_{t-1}$};
    \node[state, right=of x4] (x5) {$x_t$};

    % ===== vertical layout (tunable) =====
    \def\rowstep{0.7cm}   % gap between adjacent layers inside a round block
    \def\blockgap{1.2cm}  % gap between round blocks (full <-> partial spacing)
    \def\outgap{0.9cm}    % gap from the last matrix to the output row

    \coordinate[below=of x1] (m0); % initial external matrix
    \coordinate[below=\rowstep of m0, yshift=-0.2cm] (sw1);
    \coordinate[below=\rowstep of sw1] (m1);
    \coordinate[below=\rowstep of m1] (ac1);

    \coordinate[below=\blockgap of ac1] (s2);
    \coordinate[below=\rowstep of s2] (m2);
    \coordinate[below=\rowstep of m2] (ac2);

    \coordinate[below=\blockgap of ac2] (sw3);
    \coordinate[below=\rowstep of sw3] (m3);
    \coordinate[below=\rowstep of m3] (ac3);

    \coordinate[below=\outgap of ac3] (out);

    \coordinate (d1)  at ($(ac1)!0.5!(s2)$);     % vdots centred between blocks
    \coordinate (d2)  at ($(ac2)!0.5!(sw3)$);

    % ===== WIRES (drawn first, covered by the white layers below) =====
    \foreach \i in {1,2,4,5} {
        \draw[wire] (x\i.south)               -- ($(x\i |- ac1)+(0,-0.55)$); % initial M + block 1
        \draw[wire] ($(x\i |- s2)+(0,0.45)$)  -- ($(x\i |- ac2)+(0,-0.55)$); % block 2 (RP)
        \draw[wire] ($(x\i |- sw3)+(0,0.45)$) -- ($(x\i |- out)+(0,0.25)$);  % block 3
    }

    % ===== LAYERS + S-BOXES (cover the wires) =====
    % initial external matrix
    \node[layer={m0}{$\Mext$}] {};

    % block 1 (R_f1): wide S' boxes, M_ext, AddRC
    \node[sboxwide] at ($(x1 |- sw1)!0.5!(x2 |- sw1)$) {$S'$};
    \node[sboxwide] at ($(x4 |- sw1)!0.5!(x5 |- sw1)$) {$S'$};
    \node[empty]    at (x3 |- sw1) {\(\dots\)};
    \node[layer={m1}{$\Mext$}] {};
    \node[layer={ac1}{\AddRC}] {};

    \node[empty] at (x3 |- d1) {\(\vdots\)};

    % block 2 (RP): single S-box (last column), M_int, AddRC
    \node[sbox]  at (x5 |- s2) {$x^\alpha$};
    \node[empty] at (x3 |- s2) {\(\dots\)};
    \node[layer={m2}{$\Mint$}] {};
    \node[layer={ac2}{\AddRC}] {};

    \node[empty] at (x3 |- d2) {\(\vdots\)};

    % block 3 (R_f2): wide S' boxes, M_ext, AddRC
    \node[sboxwide] at ($(x1 |- sw3)!0.5!(x2 |- sw3)$) {$S'$};
    \node[sboxwide] at ($(x4 |- sw3)!0.5!(x5 |- sw3)$) {$S'$};
    \node[empty]    at (x3 |- sw3) {\(\dots\)};
    \node[layer={m3}{$\Mext$}] {};
    \node[layer={ac3}{\AddRC}] {};

    % ===== output state =====
    \node[state] at (x1 |- out) {$y_1$};
    \node[state] at (x2 |- out) {$y_2$};
    \node[state] at (x3 |- out) {\ldots};
    \node[state] at (x4 |- out) {$y_{t-1}$};
    \node[state] at (x5 |- out) {$y_t$};

    % ===== round-count braces (right side) =====
    \foreach \top/\bot/\lab in {sw1/ac1/\R_{\mathsf{f}_1}, s2/ac2/\RP, sw3/ac3/\R_{\mathsf{f}_2}} {
        \vbrace[raise=15pt, above=-0.25cm, below=-0.25cm]{x5 |- \bot}{x5 |- \top}{x5};
        \node[gray, rotate=-90] at ($(x5 |- \top)!0.5!(x5 |- \bot)+(0.75,0)$)
            {\footnotesize $\lab$ rounds};
    }

\end{tikzpicture}
